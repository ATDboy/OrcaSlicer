cmake_minimum_required(VERSION 3.13)
project(OrcaBrickPreviewRenderTest LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This optional GPU test uses a Windows WGL context.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(ORCA "${CMAKE_CURRENT_LIST_DIR}/../..")
add_subdirectory("${ORCA}/src/glad" glad)
add_subdirectory("${ORCA}/src/libvgcode" libvgcode)
add_executable(render_test render_test.cpp)
target_include_directories(render_test PRIVATE "${ORCA}/src")
target_link_libraries(render_test PRIVATE libvgcode opengl32 gdi32 user32)
